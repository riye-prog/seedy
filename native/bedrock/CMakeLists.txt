cmake_minimum_required(VERSION 3.25)
project(seedycrack LANGUAGES CXX)
option(SEEDY_OPENMP "Use multiple solver threads" ON)

if(NOT SEEDY_PLATFORM)
    message(FATAL_ERROR "SEEDY_PLATFORM is required")
endif()

set(BEDROCK_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../vendor/bedrockseed/src/main/c")
find_package(Python3 COMPONENTS Interpreter REQUIRED)
set(SEEDY_BLOCKS "${CMAKE_CURRENT_BINARY_DIR}/generated/blocks.cpp")
set(SEEDY_XSINV "${CMAKE_CURRENT_BINARY_DIR}/generated/xsinv.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${BEDROCK_SOURCE}/blocks.cpp" "${BEDROCK_SOURCE}/xsinv.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/prepare_blocks.py" "${CMAKE_CURRENT_SOURCE_DIR}/prepare_source.py")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/prepare_blocks.py" "${BEDROCK_SOURCE}/blocks.cpp" "${SEEDY_BLOCKS}" RESULT_VARIABLE SEEDY_BLOCK_GENERATION_STATUS ERROR_VARIABLE SEEDY_BLOCK_GENERATION_ERROR)
if(NOT SEEDY_BLOCK_GENERATION_STATUS EQUAL 0)
    message(FATAL_ERROR "${SEEDY_BLOCK_GENERATION_ERROR}")
endif()
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/prepare_source.py" "${BEDROCK_SOURCE}/xsinv.cpp" "${SEEDY_XSINV}" RESULT_VARIABLE SEEDY_GENERATION_STATUS ERROR_VARIABLE SEEDY_GENERATION_ERROR)
if(NOT SEEDY_GENERATION_STATUS EQUAL 0)
    message(FATAL_ERROR "${SEEDY_GENERATION_ERROR}")
endif()
find_path(SEEDY_JNI_INCLUDE jni.h HINTS "${JAVA_HOME}/include" "$ENV{JAVA_HOME}/include" REQUIRED)
if(WIN32)
    find_path(SEEDY_JNI_PLATFORM jni_md.h HINTS "${JAVA_HOME}/include/win32" "$ENV{JAVA_HOME}/include/win32" REQUIRED)
else()
    find_path(SEEDY_JNI_PLATFORM jni_md.h HINTS "${JAVA_HOME}/include/darwin" "$ENV{JAVA_HOME}/include/darwin" REQUIRED)
endif()
find_path(SEEDY_Z3_INCLUDE z3++.h HINTS "${Z3_ROOT}/include" REQUIRED)
find_library(SEEDY_Z3_LIBRARY NAMES z3 libz3 HINTS "${Z3_ROOT}/lib" "${Z3_ROOT}/bin" REQUIRED)
add_library(seedycrack SHARED "${SEEDY_BLOCKS}" java_helper.cpp)
target_compile_features(seedycrack PRIVATE cxx_std_23)
target_compile_options(seedycrack PRIVATE -O3 -fPIC)
if(SEEDY_OPENMP)
    target_compile_options(seedycrack PRIVATE -fopenmp)
    target_link_options(seedycrack PRIVATE -fopenmp)
endif()
target_include_directories(seedycrack PRIVATE "${BEDROCK_SOURCE}" ${SEEDY_JNI_INCLUDE} ${SEEDY_JNI_PLATFORM} ${SEEDY_Z3_INCLUDE})
target_link_libraries(seedycrack PRIVATE ${SEEDY_Z3_LIBRARY})
if(SEEDY_PLATFORM STREQUAL "macos-arm64")
    add_library(seedycrack_baseline OBJECT "${SEEDY_XSINV}")
    target_compile_features(seedycrack_baseline PRIVATE cxx_std_23)
    target_include_directories(seedycrack_baseline PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_compile_definitions(seedycrack_baseline PRIVATE KERNEL_NAME=crackFP_baseline)
    target_compile_options(seedycrack_baseline PRIVATE -O3 -fPIC)
    if(SEEDY_OPENMP)
        target_compile_options(seedycrack_baseline PRIVATE -fopenmp)
    endif()
    target_sources(seedycrack PRIVATE $<TARGET_OBJECTS:seedycrack_baseline>)
else()
    foreach(variant baseline avx2 avx512)
        add_library(seedycrack_${variant} OBJECT "${SEEDY_XSINV}")
        target_compile_features(seedycrack_${variant} PRIVATE cxx_std_23)
        target_include_directories(seedycrack_${variant} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
        target_compile_definitions(seedycrack_${variant} PRIVATE KERNEL_NAME=crackFP_${variant})
        target_compile_options(seedycrack_${variant} PRIVATE -O3 -fPIC)
        if(APPLE AND CMAKE_CXX_COMPILER_ID MATCHES "Clang")
            target_compile_options(seedycrack_${variant} PRIVATE -include "${CMAKE_CURRENT_SOURCE_DIR}/simd_compat.h")
        endif()
        if(SEEDY_OPENMP)
            target_compile_options(seedycrack_${variant} PRIVATE -fopenmp)
        endif()
        if(variant STREQUAL "avx2")
            target_compile_options(seedycrack_${variant} PRIVATE -mavx2)
        elseif(variant STREQUAL "avx512")
            target_compile_options(seedycrack_${variant} PRIVATE -mavx512f)
        endif()
        target_sources(seedycrack PRIVATE $<TARGET_OBJECTS:seedycrack_${variant}>)
    endforeach()
endif()
if(APPLE)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
        target_link_options(seedycrack PRIVATE -static-libgcc -static-libstdc++)
    endif()
    set_target_properties(seedycrack PROPERTIES INSTALL_NAME_DIR "@rpath" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../dist/${SEEDY_PLATFORM}")
else()
    set_target_properties(seedycrack PROPERTIES PREFIX "" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../dist/${SEEDY_PLATFORM}")
endif()
