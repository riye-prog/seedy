cmake_minimum_required(VERSION 3.24)
project(seedy LANGUAGES C CXX)
find_path(SEEDY_JNI_INCLUDE jni.h HINTS "${JAVA_HOME}/include" "$ENV{JAVA_HOME}/include" REQUIRED)
if(APPLE)
    set(SEEDY_JNI_PLATFORM darwin)
elseif(WIN32)
    set(SEEDY_JNI_PLATFORM win32)
else()
    message(FATAL_ERROR "Supported platforms: macOS and Windows")
endif()
find_path(SEEDY_JNI_PLATFORM_INCLUDE jni_md.h HINTS "${SEEDY_JNI_INCLUDE}/${SEEDY_JNI_PLATFORM}" REQUIRED)
include(FetchContent)
FetchContent_Declare(imgui GIT_REPOSITORY https://github.com/ocornut/imgui.git GIT_TAG v1.91.9b GIT_SHALLOW TRUE)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz)
FetchContent_Declare(vulkan_headers GIT_REPOSITORY https://github.com/KhronosGroup/Vulkan-Headers.git GIT_TAG v1.4.304 GIT_SHALLOW TRUE)
FetchContent_Declare(volk GIT_REPOSITORY https://github.com/zeux/volk.git GIT_TAG vulkan-sdk-1.4.304.0 GIT_SHALLOW TRUE)
set(VOLK_PULL_IN_VULKAN OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(imgui json vulkan_headers)
set(VULKAN_HEADERS_INSTALL_DIR "${vulkan_headers_SOURCE_DIR}" CACHE PATH "" FORCE)
FetchContent_MakeAvailable(volk)
target_link_libraries(volk PUBLIC Vulkan::Headers)
file(READ "${imgui_SOURCE_DIR}/misc/fonts/Roboto-Medium.ttf" SEEDY_FONT HEX)
string(REGEX REPLACE "(..)" "0x\\1," SEEDY_FONT_BYTES "${SEEDY_FONT}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/font_data.h" "static const unsigned char font_data[] = {${SEEDY_FONT_BYTES}};")
add_library(seedy SHARED overlay.cpp vulkan_renderer.cpp ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_vulkan.cpp)
target_include_directories(seedy PRIVATE ${SEEDY_JNI_INCLUDE} ${SEEDY_JNI_PLATFORM_INCLUDE} ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends ${CMAKE_CURRENT_BINARY_DIR})
target_compile_definitions(seedy PRIVATE IMGUI_IMPL_VULKAN_USE_VOLK)
target_link_libraries(seedy PRIVATE nlohmann_json::nlohmann_json volk Vulkan::Headers)
if(APPLE)
    enable_language(OBJCXX)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    set(SEEDY_METAL_BACKEND "${CMAKE_CURRENT_BINARY_DIR}/imgui_seedy_metal.mm")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/prepare_metal.py")
    execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/prepare_metal.py" "${imgui_SOURCE_DIR}/backends/imgui_impl_metal.mm" "${SEEDY_METAL_BACKEND}" COMMAND_ERROR_IS_FATAL ANY)
    target_sources(seedy PRIVATE metal_renderer.mm "${SEEDY_METAL_BACKEND}")
    set_source_files_properties(metal_renderer.mm "${SEEDY_METAL_BACKEND}" PROPERTIES COMPILE_OPTIONS "-fobjc-arc")
    target_link_libraries(seedy PRIVATE "-framework Metal" "-framework Foundation" "-framework QuartzCore")
    set_target_properties(seedy PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/dist/macos-universal")
    find_program(SEEDY_CODESIGN codesign REQUIRED)
    add_custom_command(TARGET seedy POST_BUILD COMMAND "${SEEDY_CODESIGN}" --force --sign - "$<TARGET_FILE:seedy>" VERBATIM)
else()
    set_target_properties(seedy PROPERTIES PREFIX "" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/dist/windows-x86_64" RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_SOURCE_DIR}/dist/windows-x86_64")
endif()
target_compile_features(seedy PRIVATE cxx_std_17)
include(CTest)
if(BUILD_TESTING AND APPLE)
    add_executable(seedy_interaction_test tests/interaction.cpp ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp)
    target_include_directories(seedy_interaction_test PRIVATE ${SEEDY_JNI_INCLUDE} ${SEEDY_JNI_PLATFORM_INCLUDE} ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends ${CMAKE_CURRENT_BINARY_DIR})
    target_compile_definitions(seedy_interaction_test PRIVATE IMGUI_ENABLE_TEST_ENGINE SEEDY_HEADLESS_TEST)
    target_compile_features(seedy_interaction_test PRIVATE cxx_std_17)
    target_link_libraries(seedy_interaction_test PRIVATE nlohmann_json::nlohmann_json)
    add_test(NAME native_panel_interaction COMMAND seedy_interaction_test)
endif()

option(SEEDY_VULKAN_BENCHMARK "Build the command-line Vulkan GPU benchmark" OFF)
if(SEEDY_VULKAN_BENCHMARK AND APPLE)
    foreach(mode current baseline)
        if(mode STREQUAL "baseline")
            if(NOT EXISTS "${SEEDY_BASELINE_RENDERER}")
                continue()
            endif()
            set(renderer_source "${SEEDY_BASELINE_RENDERER}")
        else()
            set(renderer_source vulkan_renderer.cpp)
        endif()
        add_executable(seedy_vulkan_${mode} tests/vulkan_benchmark.cpp "${renderer_source}" ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_vulkan.cpp)
        target_include_directories(seedy_vulkan_${mode} PRIVATE ${SEEDY_JNI_INCLUDE} ${SEEDY_JNI_PLATFORM_INCLUDE} ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends ${CMAKE_CURRENT_BINARY_DIR} ${CMAKE_CURRENT_SOURCE_DIR})
        target_compile_definitions(seedy_vulkan_${mode} PRIVATE IMGUI_IMPL_VULKAN_USE_VOLK)
        if(mode STREQUAL "baseline")
            target_compile_definitions(seedy_vulkan_${mode} PRIVATE SEEDY_LEGACY_RENDERER)
        endif()
        target_compile_features(seedy_vulkan_${mode} PRIVATE cxx_std_17)
        target_link_libraries(seedy_vulkan_${mode} PRIVATE nlohmann_json::nlohmann_json volk Vulkan::Headers)
    endforeach()
endif()
